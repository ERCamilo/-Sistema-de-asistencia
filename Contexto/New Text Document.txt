-- PASO 2: CREAR TABLAS
-- ═══════════════════════════════════════════════════════════════════════

-- 1. TABLA DE PERFILES DE USUARIO
CREATE TABLE profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE,
  email TEXT,
  full_name TEXT,
  role TEXT DEFAULT 'user',
  company_id UUID,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  PRIMARY KEY (id)
);

-- 2. TABLA DE POSICIONES (ACTUALIZADA)
CREATE TABLE positions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  color TEXT,
  hourly_rate DECIMAL,
  working_days INTEGER[],
  leader_id UUID,  -- ✅ NUEVO: Líder de esta posición
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- ⚠️ La foreign key de leader_id se agrega después de crear employees

-- 3. TABLA DE EMPLEADOS
CREATE TABLE employees (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  number TEXT NOT NULL,
  name TEXT NOT NULL,
  positions UUID[],
  position_salaries JSONB,      -- { "uuid-pos-1": 25000, "uuid-pos-2": 30000 }
  custom_working_days JSONB,    -- { "uuid-pos-1": [1,2,3,4,5], "uuid-pos-2": [6,0] }
  hire_date DATE,
  phone TEXT,
  email TEXT,
  notes TEXT,
  active BOOLEAN DEFAULT true,
  last_payment_date DATE,
  status_history JSONB,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(user_id, number)
);

-- Ahora sí podemos agregar la foreign key de leader_id
ALTER TABLE positions 
  ADD CONSTRAINT fk_leader 
  FOREIGN KEY (leader_id) REFERENCES employees(id) ON DELETE SET NULL;

-- 4. TABLA DE ASISTENCIA (ACTUALIZADA)
CREATE TABLE attendance (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  employee_id UUID REFERENCES employees(id) ON DELETE CASCADE,
  date DATE NOT NULL,
  present BOOLEAN,
  hours_worked DECIMAL,          -- Total de horas trabajadas
  position_id UUID REFERENCES positions(id),  -- Posición principal (para compatibilidad)
  notes TEXT,                    -- ✅ NUEVO: Notas del día
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(employee_id, date)
);

-- 5. TABLA DE HORAS POR POSICIÓN (NUEVA)
-- Esta tabla permite que un empleado trabaje múltiples posiciones el mismo día
CREATE TABLE attendance_positions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  attendance_id UUID REFERENCES attendance(id) ON DELETE CASCADE,
  position_id UUID REFERENCES positions(id),
  hours DECIMAL NOT NULL,
  overtime_hours DECIMAL DEFAULT 0,
  created_at TIMESTAMP DEFAULT NOW()
);

-- 6. TABLA DE DEDUCCIONES
CREATE TABLE deductions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  employee_id UUID REFERENCES employees(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  percentage DECIMAL,
  applies_to_next BOOLEAN DEFAULT false,
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMP DEFAULT NOW()
);

-- 7. TABLA DE CONFIGURACIÓN
CREATE TABLE settings (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE UNIQUE,
  company_name TEXT,
  regular_hours_per_day INTEGER DEFAULT 8,
  overtime_factor DECIMAL DEFAULT 1.5,
  holiday_factor DECIMAL DEFAULT 2,
  default_deduction_percentage DECIMAL DEFAULT 2,
  holidays DATE[],
  last_payment_date DATE,
  next_payment_date DATE,
  calendar_marker_mode TEXT DEFAULT 'holiday',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- PASO 5: CREAR POLICIES (SEGURIDAD)
-- ═══════════════════════════════════════════════════════════════════════

-- PROFILES
CREATE POLICY "Users can view own profile" 
  ON profiles FOR SELECT
  USING (auth.uid() = id);

CREATE POLICY "Users can update own profile" 
  ON profiles FOR UPDATE
  USING (auth.uid() = id);

-- POSITIONS
CREATE POLICY "Users see own positions" 
  ON positions FOR ALL
  USING (user_id = auth.uid());

-- EMPLOYEES
CREATE POLICY "Users see own employees" 
  ON employees FOR ALL
  USING (user_id = auth.uid());

-- ATTENDANCE
CREATE POLICY "Users see own attendance" 
  ON attendance FOR ALL
  USING (user_id = auth.uid());

-- ATTENDANCE_POSITIONS
CREATE POLICY "Users see own attendance positions" 
  ON attendance_positions FOR ALL
  USING (
    EXISTS (
      SELECT 1 FROM attendance a 
      WHERE a.id = attendance_positions.attendance_id 
      AND a.user_id = auth.uid()
    )
  );

-- DEDUCTIONS
CREATE POLICY "Users see own deductions" 
  ON deductions FOR ALL
  USING (user_id = auth.uid());

-- SETTINGS
CREATE POLICY "Users see own settings" 
  ON settings FOR ALL
  USING (user_id = auth.uid());

-- Función para crear perfil automáticamente
CREATE OR REPLACE FUNCTION public.handle_new_user() 
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email, full_name)
  VALUES (new.id, new.email, new.raw_user_meta_data->>'full_name');
  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger cuando se registra un usuario
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- Función auxiliar para settings
CREATE OR REPLACE FUNCTION get_or_create_settings(p_user_id UUID)
RETURNS TABLE (
  id UUID,
  user_id UUID,
  company_name TEXT,
  regular_hours_per_day INTEGER,
  overtime_factor DECIMAL,
  holiday_factor DECIMAL,
  default_deduction_percentage DECIMAL,
  holidays DATE[],
  last_payment_date DATE,
  next_payment_date DATE,
  calendar_marker_mode TEXT,
  created_at TIMESTAMP,
  updated_at TIMESTAMP
) AS $$
BEGIN
  RETURN QUERY
  SELECT s.* FROM settings s WHERE s.user_id = p_user_id;
  
  IF NOT FOUND THEN
    INSERT INTO settings (user_id) VALUES (p_user_id);
    RETURN QUERY
    SELECT s.* FROM settings s WHERE s.user_id = p_user_id;
  END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
